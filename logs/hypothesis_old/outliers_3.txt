# Hypothesis 10.
description = "The last name of the passengers does not correlate with survival."
def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency
    
    # Extract last name
    dataset['LastName'] = dataset['Name'].apply(lambda x: x.split(',')[0])

    # Analyze those with shared last names
    shared_last_names = dataset.groupby('LastName')['Survived'].agg(lambda x: x.mode()[0] if not x.mode().empty else 0)
    
    # Create a contingency table
    contingency_table = pd.crosstab(shared_last_names, dataset['Survived'])
    
    # Perform Chi-Squared test
    chi2, p, _, _ = chi2_contingency(contingency_table)
    
    # Check if p-value is greater than 0.05
    result = int(p > 0.05)
    return result
