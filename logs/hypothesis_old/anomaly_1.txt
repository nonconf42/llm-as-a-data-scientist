description = "Test if there is a significant difference in survival rates between different ticket classes using the Chi-Square test."
def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency

    contingency_table = pd.crosstab(dataset['Pclass'], dataset['Survived'])
    chi2, p, dof, expected = chi2_contingency(contingency_table)

    return result

description = "Test if high.ticket.fare is associated with a significant difference in survival rates using T-test."
def test(dataset):
    from scipy.stats import ttest_ind

    survived = dataset[dataset['Survived'] == 1]['Fare'].dropna()
    not_survived = dataset[dataset['Survived'] == 0]['Fare'].dropna()
    
    t_stat, p = ttest_ind(survived, not_survived)

    result = 1 if p < 0.05 else 0
    return result

description = "Analyze the survival trend over fare amounts using a linear regression and test for significance."
def test(dataset):
    import statsmodels.api as sm
    import numpy as np

    dataset = dataset[['Survived', 'Fare']].dropna()
    X = sm.add_constant(dataset['Fare'])
    y = dataset['Survived']
    model = sm.OLS(y, X)
    results = model.fit()

    result = 1 if results.pvalues['Fare'] < 0.05 else 0
    return result

description = "Evaluate if the number of siblings/spouses aboard significantly influences fare paid through correlation analysis."

    import numpy as np

    corr, p = pearsonr(fare_sibsp_data['Fare'], fare_sibsp_data['SibSp'])

    result = 1 if abs(corr) > 0.3 and p < 0.05 else 0
    return result
