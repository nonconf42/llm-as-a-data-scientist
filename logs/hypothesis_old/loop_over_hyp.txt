def test(dataset):
    import pandas as pd
    import statsmodels.api as sm

    dataset = dataset.dropna(subset=['Cabin'])  # Handle missing data
    dataset['CabinType'] = dataset['Cabin'].apply(lambda x: x[0])
    X = pd.get_dummies(dataset['CabinType'], drop_first=True)
    X = sm.add_constant(X)
    y = dataset['Survived']
    log_reg = sm.Logit(y, X).fit()

    p_values = log_reg.pvalues[1:]  # Exclude intercept
    alpha = 0.05
    result = 1 if any(p < alpha for p in p_values) else 0
    return result

description = "Investigate if latent variables derived from family size influence the relationship between class and survival."
COT = "Test latent mediation by using mediation analysis with perceived family size as mediator between 'Pclass' and 'Survived'."
def test(dataset):
    import pandas as pd
    from statsmodels.formula.api import ols
    from statsmodels.stats.mediation import Mediation

    def calc_family_size(df):
        df['FamilySize'] = df['SibSp'] + df['Parch']
        return df[['Pclass', 'FamilySize', 'Survived']]
    
    df = calc_family_size(dataset).dropna()
    
    # Mediation Model
    model_mediator = ols("FamilySize ~ Pclass", data=df).fit()
    model_outcome = ols("Survived ~ Pclass + FamilySize", data=df).fit()
    
    med = Mediation(model_outcome, model_mediator, 'FamilySize', 'Pclass')
    med_res = med.fit()
    result = int(med_res.pvalue() < 0.05)
    return result
