description = "First class passengers are more likely to survive compared to second and third class passengers."
def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency

    # Calculate contingency table
    survival_by_class = pd.crosstab(dataset['Pclass'], dataset['Survived'])
    _, p_value, _, _ = chi2_contingency(survival_by_class)
    
    # p-value of less than 0.05 indicates strong evidence to accept the hypothesis
    result = 1 if p_value < 0.05 else 0
    return result

description = "Passengers in a certain cabin area may have had different survival rates."
def test(dataset):
    import pandas as pd

    # Fill missing cabin values
    dataset['Cabin_Filled'] = dataset['Cabin'].fillna('Unknown')
    dataset['Cabin_Area'] = dataset['Cabin_Filled'].astype(str).str[0]  # First letter as cabin area
    try:
        from sklearn.feature_selection import mutual_info_classif
        X = pd.get_dummies(dataset['Cabin_Area'])
        y = dataset['Survived']
        result = mutual_info_classif(X, y, discrete_features=True).max() > 0.01
    except ImportError:
        # If sklearn is not present, fall back to contingency test
        from scipy.stats import chi2_contingency
        survival_by_cabin = pd.crosstab(dataset['Cabin_Area'], dataset['Survived'])
        _, p_value, _, _ = chi2_contingency(survival_by_cabin)
        result = p_value < 0.05
    
    return int(result)

# Hypothesis 9.
description = "There are distinct clusters of passengers with different survival rates based on age and fare."
def test(dataset):
    import numpy as np
    from sklearn.cluster import KMeans
    from sklearn.preprocessing import StandardScaler
    from sklearn.metrics import silhouette_score

    # Preprocess data
    df = dataset[['Age', 'Fare']].dropna()
    X = StandardScaler().fit_transform(df)
    
    # Apply KMeans and evaluate silhouette score
    kmeans = KMeans(n_clusters=3, random_state=42).fit(X)
    score = silhouette_score(X, kmeans.labels_)
    return 1 if score > 0.5 else 0
