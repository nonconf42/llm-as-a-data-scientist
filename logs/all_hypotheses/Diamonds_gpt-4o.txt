        # Fit regression model

    p_value = model.pvalues['log_carat']  # Get p-value for carat coefficient

    # Return 1 if significant

    return result

    carat_premium_ratio = mean_large / mean_medium

    from statsmodels.formula.api import ols
    def analyze_carat_price(data):

        model = ols('price ~ carat', data=data).fit()
        predictions = model.predict(data['carat'])
        diffs = data['price'] - predictions
        high_diff = diffs[data['group'] == 'High']
        low_diff = diffs[data['group'] == 'Low']
        t_stat, p_value = ttest_rel(high_diff, low_diff)
        return p_value < 0.05 and np.mean(high_diff) > np.mean(low_diff)
    result = analyze_carat_price(dataset)
    return result
