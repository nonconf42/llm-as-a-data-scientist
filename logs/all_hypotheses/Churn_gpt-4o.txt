def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency
    
    # Step 1: Split the data into two groups based on age.
    young_customers = dataset[dataset['Age'] <= 40]
    older_customers = dataset[dataset['Age'] > 40]
    
    # Step 2: Calculate retention for each group.
    young_ret = young_customers['Exited'].value_counts().get(0, 0)
    old_ret = older_customers['Exited'].value_counts().get(0, 0)
    
    young_total = len(young_customers)
    old_total = len(older_customers)
    
    # Chi-squared test setup:
    contingency_table = [
        [young_ret, young_total - young_ret],
        [old_ret, old_total - old_ret]
    ]
    
    # Step 3: Perform chi-squared test.
    chi2, p, _, _ = chi2_contingency(contingency_table)
    
    # Step 4: Accept hypothesis if p-value < 0.05.
    result = int(p < 0.05)
    
    return result
