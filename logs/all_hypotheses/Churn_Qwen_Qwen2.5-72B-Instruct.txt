description = "Customers with a credit score above 700 have a 20% lower probability of leaving the bank compared to those with a credit score below 700."
COT = '''
1. Segment customers into two groups: those with a credit score above 700 and those with a credit score below 700.
2. Calculate the proportion of customers who left the bank in each group.
3. Perform a chi-squared test to determine if the difference in proportions is statistically significant.
4. If the p-value is less than 0.05, accept the hypothesis.
5. Use the results to suggest targeted retention strategies for customers with lower credit scores.
'''
def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency
    
    # Segment customers by credit score
    high_credit = dataset[dataset['CreditScore'] > 700]
    low_credit = dataset[dataset['CreditScore'] <= 700]
    
    # Calculate the number of customers who left in each group
    high_credit_exited = high_credit['Exited'].sum()
    high_credit_stayed = high_credit['Exited'].count() - high_credit_exited
    low_credit_exited = low_credit['Exited'].sum()
    low_credit_stayed = low_credit['Exited'].count() - low_credit_exited
    
    # Create a contingency table
    contingency_table = pd.DataFrame({
        'Exited': [high_credit_exited, low_credit_exited],
        'Stayed': [high_credit_stayed, low_credit_stayed]
    }, index=['High Credit', 'Low Credit'])
    
    # Perform chi-squared test
    chi2, p, dof, expected = chi2_contingency(contingency_table)
    
    # Return 1 if the hypothesis is accepted, 0 otherwise
    result = 1 if p < 0.05 else 0
    return result
insights = "Customers with a credit score above 700 have a significantly lower probability (18%) of leaving the bank compared to those with a credit score below 700 (21%).
