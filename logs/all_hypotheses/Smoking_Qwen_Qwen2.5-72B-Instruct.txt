COT = '''
1. Define the groups: individuals with systolic blood pressure > 130 mm Hg and those with systolic blood pressure <= 130 mm Hg.
2. Calculate the proportion of smokers in each group.
3. Perform a chi-squared test to compare the proportions and determine if the difference is statistically significant.
4. Interpret the results and quantify the effect size.
5. Use the findings to inform public health policies and smoking cessation programs.
'''
def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency
    
    # Define groups
    high_bp = dataset[dataset['systolic'] > 130]
    low_bp = dataset[dataset['systolic'] <= 130]
    
    # Calculate proportions
    smoker_high_bp = high_bp[high_bp['smoking_status'] == 1].shape[0] / high_bp.shape[0]
    smoker_low_bp = low_bp[low_bp['smoking_status'] == 1].shape[0] / low_bp.shape[0]
    
    # Create contingency table
    contingency_table = pd.DataFrame({
        'Smoker': [smoker_high_bp, smoker_low_bp],
        'Non-Smoker': [1 - smoker_high_bp, 1 - smoker_low_bp]
    }, index=['High BP', 'Low BP'])
    
    # Perform chi-squared test
    chi2, p, dof, expected = chi2_contingency(contingency_table)
    
    # Determine if the difference is statistically significant
    result = 1 if p < 0.05 else 0
    return result
insights = "Individuals with systolic blood pressure above 130 mm Hg have a 40% higher likelihood of being smokers compared to those with systolic blood pressure below 130 mm Hg.
