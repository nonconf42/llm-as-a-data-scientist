5. Structure the function to accept a DataFrame and perform the analysis dynamically.

'''
def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency

    # Define blood pressure thresholds
    high_bp = dataset[dataset['systolic'] > 140]
    low_bp = dataset[dataset['systolic'] < 120]

    # Count smokers and non-smokers for each group

    high_bp_total = len(high_bp)

    # Calculate average eyesight for each individual
    dataset['average_eyesight'] = dataset[['eyesight(left)', 'eyesight(right)']].mean(axis=1)

    poor_eyesight = dataset[dataset['average_eyesight'] > 1.0]
    good_eyesight = dataset[dataset['average_eyesight'] <= 1.0]
    
    # Calculate smoking rates
    poor_eyesight_smokers = poor_eyesight['smoking_status'].mean()
    good_eyesight_smokers = good_eyesight['smoking_status'].mean()

    contingency_table = [[poor_eyesight['smoking_status'].sum(), len(poor_eyesight) - poor_eyesight['smoking_status'].sum()],

    # Perform chi-squared test
    chi2, p, _, _ = stats.chi2_contingency(contingency_table)
    
    # Determine if there's a significant difference
    result = 1 if p < 0.05 and poor_eyesight_smokers > good_eyesight_smokers else 0
    return result
