    import pandas as pd
    import numpy as np
    from scipy.stats import pearsonr
    assets = dataset['Assets']
    liabilities = dataset['Liabilities']
    correlation_coefficient, _ = pearsonr(assets, liabilities)
    result = 1 if correlation_coefficient >= 0.9 else 0
    return result
