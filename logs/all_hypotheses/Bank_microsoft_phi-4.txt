    # Gather subsequent net income variance for top 20% and bottom 80%

    next_quarter_variances_low = [dataset[dataset['Quarter'] == q + 1]['Net_Income'].var() for q in other_quarters]
    
    # Perform an F-test for equality of variances
    f_stat, p_value = f.criterion(next_quarter_variances_high, next_quarter_variances_low)

def test_hypothesis_5(dataset):
    import pandas as pd
    from scipy.stats import ttest_ind
    
    # Convert 'Date' to datetime and extract quarters
    dataset['Date'] = pd.to_datetime(dataset['Date'])
    dataset['Quarter'] = dataset['Date'].dt.to_period('Q')

    top_cutoff = pd.qcut(dataset['Dividend_Payout'], q=0.8, labels=False).max()

    # Cash Flow in subsequent quarters
    subsequent_cash_flows = []
    for q in high_payout_quarters:
        subsequent_cash_flow = dataset[dataset['Quarter'] == q + 1]['Cash_Flow'].values
        if len(subsequent_cash_flow) > 0:
            subsequent_cash_flows.extend(subsequent_cash_flow)
    
    # Cash Flow in other quarters
    other_cash_flows = dataset[~dataset['Quarter'].isin(high_payout_quarters)]['Quarter'].shift(-1).dropna()
    other_cash_flows = dataset[dataset['Quarter'].isin(other_cash_flows)]['Cash_Flow'].values
    
    # Perform t-test to compare subsequent and other cash flows
    t_stat, p_value = ttest_ind(subsequent_cash_flows, other_cash_flows, equal_var=False)
    result = 1 if p_value < 0.05 else 0
    
    return result
