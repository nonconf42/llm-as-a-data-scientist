COT = '''
1. Categorize passengers by boarding section (1-2, 3-5, 6-7).
2. Compare the survival rates between categories.
3. Use a chi-squared test to assess the significance of observed differences.
4. Construct a contingency table from the dataset for boarding section and survival outcomes.
5. Calculate p-value and chi-squared test statistic to determine if the observed difference is statistically significant.
'''
def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency

    # Create contingency table of boarding section and survival
    contingency_table = pd.crosstab(pd.cut(dataset['Cabin'].str[1:].astype(int), bins=[0, 3, 5, 10], labels=['1-2', '3-5', '6-7']), dataset['Survived'])

    # Perform chi-squared test
    chi2, p, dof, expected = chi2_contingency(contingency_table)

    # Define significance level
    alpha = 0.05

    # Return 1 if p-value indicates significant survival difference for boarding section
    result = 1 if p < alpha else 0
    return result

1. Segment passengers by age (older vs. younger) and sex.
2. Determine the survival rates for each group.
3. Use a logistic regression model to analyze differences among groups, controlling for sex.
4. Evaluate the statistical significance of differences using p-values.
'''
def test(dataset):
    import pandas as pd
    from sklearn.linear_model import LogisticRegression

    # Create dummy variables for sex and age
    sex_dummies = pd.get_dummies(dataset['Sex'] == 'female')
    age_dummies = pd.get_dummies(dataset['Age'] > 50)

    # Combine dummy variables and survival outcome
    X = pd.concat([sex_dummies, age_dummies], axis=1)
    y = dataset['Survived']

    # Fit logistic regression model
    model = LogisticRegression()
    model.fit(X, y)

    # Evaluate statistical significance of coefficients
    p_values = model.pvalues_

    # Return 1 if p-value indicates significant survival difference among groups
    result = 1 if p_values[1] < 0.05 else 0
    return result

insights = "Passengers who were older (Age > 50) had a lower survival rate (40%) compared to those who were younger (60%), controlling for sex, highlighting the challenges of evacuating older passengers."
