def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency
    
    # Segments based on product category
    clothing_customers = dataset[dataset['Category'] == 'Clothing']
    other_customers = dataset[dataset['Category'] != 'Clothing']
    
    # Count promo code usage
    clothing_with_promo = clothing_customers[clothing_customers['Promo Code Used'] == 'Yes'].shape[0]
    clothing_without_promo = clothing_customers[clothing_customers['Promo Code Used'] == 'No'].shape[0]
    
    other_with_promo = other_customers[other_customers['Promo Code Used'] == 'Yes'].shape[0]
    other_without_promo = other_customers[other_customers['Promo Code Used'] == 'No'].shape[0]
    
    # Construct the contingency table and perform chi-squared test
    contingency_table = [[clothing_with_promo, clothing_without_promo],
                         [other_with_promo, other_without_promo]]
    
    chi2, p_value, _, _ = chi2_contingency(contingency_table)
    # If p-value < 0.05, we accept the hypothesis
    result = 1 if p_value < 0.05 else 0
    
    return result
