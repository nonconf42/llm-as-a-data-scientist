COT = '''
1. Isolate transactions involving accessories and segment by shipping type, particularly 'Express.'
2. Measure average review ratings for express vs standard shipments.
3. Conduct an ANOVA test to detect significant review rating differences between shipping options.
4. Choose a significance level of 0.05.
5. Script the function to perform data manipulation and execute the ANOVA test accurately.
'''
def test5(dataset):
    import pandas as pd
    from scipy.stats import f_oneway

    express_ratings = dataset[(dataset['Category'] == 'Accessories') & (dataset['Shipping Type'] == 'Express')]['Review Rating']
    standard_ratings = dataset[(dataset['Category'] == 'Accessories') & (dataset['Shipping Type'] == 'Standard')]['Review Rating']
    
    f_stat, p_value = f_oneway(express_ratings, standard_ratings)
    alpha = 0.05
    result = 1 if p_value < alpha else 0
    return result
