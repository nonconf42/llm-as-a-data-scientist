    from scipy.stats import chi2_contingency

                   [older_digital, older_total - older_digital]]
    
    # Perform chi-squared test
    chi2_stat, p_value, _, _ = chi2_contingency(contingency)
    
    # Set significance level

    # Accept the hypothesis if p-value < alpha

    return result

    from scipy.stats import chisquare

    # Filter the dataset for age group 18-30
    age_group_data = dataset[dataset['Age'].between(18, 30)]

    # Count purchases by category

    # Perform a chi-squared goodness-of-fit test
    expected_frequencies = [category_counts.mean()] * len(category_counts)
    chi2, p = chisquare(f_obs=category_counts, f_exp=expected_frequencies)

    # Set the significance level
    alpha = 0.05

    # Result: 1 if significant, 0 otherwise
    result = 1 if p < alpha else 0

    return result
