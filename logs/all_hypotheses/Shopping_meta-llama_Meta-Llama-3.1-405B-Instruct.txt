COT = '''
1. Segment customers into groups: those who purchased items during the spring season and those who purchased during other seasons.
2. Calculate the percentage of customers in each group who left a review with a 4-star or higher rating.
3. Compare the probabilities between the two groups using a statistical test (e.g., chi-squared test).
4. Assess the significance of the observed difference to confirm the hypothesis.
5. Use results to inform seasonal marketing strategies.
'''
def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency
    # Filter customers who purchased items during the spring season
    spring_df = dataset[dataset['Season'] == 'Spring']
    # Create a contingency table for the chi-squared test
    contingency_table = pd.crosstab(spring_df['Season'], spring_df['Review Rating'] >= 4)
    # Perform the chi-squared test
    chi2, p, dof, expected = chi2_contingency(contingency_table)
    # Return 1 if the hypothesis is accepted, 0 otherwise
    return 1 if p < 0.05 else 0
