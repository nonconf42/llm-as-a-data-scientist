def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency

    # Categorize based on activity status
    active_members = dataset[dataset['IsActiveMember'] == 1]
    inactive_members = dataset[dataset['IsActiveMember'] == 0]

    # Calculate exit counts
    active_left = active_members['Exited'].value_counts().get(1, 0)
    active_retained = active_members['Exited'].value_counts().get(0, 0)

    inactive_left = inactive_members['Exited'].value_counts().get(1, 0)
    inactive_retained = inactive_members['Exited'].value_counts().get(0, 0)

    # Create contingency table
    contingency_table = [[active_retained, active_left], [inactive_retained, inactive_left]]

    # Perform chi-squared test
    chi2, p, _, _ = chi2_contingency(contingency_table)

    # Accept hypothesis if p-value < 0.05
    result = int(p < 0.05)

    return result
