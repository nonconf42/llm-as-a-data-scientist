# Hypothesis 1.
description = "Individuals with higher systolic blood pressure (>130 mmHg) are more likely to be smokers, with a 25% higher odds ratio compared to those with normal blood pressure."
COT = '''
1. Identify the variables of interest: systolic blood pressure and smoking status.
2. Define the groups: individuals with high systolic blood pressure (>130 mmHg) and those with normal blood pressure.
3. Calculate the odds ratio of being a smoker for individuals with high systolic blood pressure compared to those with normal blood pressure.
4. Perform a statistical test (e.g., logistic regression) to determine the significance of the odds ratio.
5. Interpret the results and quantify the effect size of high systolic blood pressure on smoking status.
'''
def test(dataset):
    import pandas as pd
    import numpy as np
    from scipy.stats import logistic
    # Define the groups
    high_bp = dataset[dataset['systolic'] > 130]
    normal_bp = dataset[dataset['systolic'] <= 130]
    # Calculate the odds ratio
    odds_ratio = (high_bp['smoking_status'].mean() / (1 - high_bp['smoking_status'].mean())) / (normal_bp['smoking_status'].mean() / (1 - normal_bp['smoking_status'].mean()))
    # Perform logistic regression
    X = dataset[['systolic']]
    y = dataset['smoking_status']
    model = logistic.LogisticRegression()
    model.fit(X, y)
    result = model.predict(X)
    return result
insights = "Individuals with high systolic blood pressure (>130 mmHg) are 25% more likely to be smokers, indicating a potential link between cardiovascular health and smoking habits. Healthcare providers can use this insight to target smoking cessation programs for individuals with high blood pressure."

description = "Smokers have lower levels of HDL cholesterol, with an average difference of 10 mg/dL compared to non-smokers."
COT = '''
1. Identify the variables of interest: HDL cholesterol and smoking status.
2. Define the groups: smokers and non-smokers.
3. Calculate the average HDL cholesterol levels for smokers and non-smokers.
4. Perform a statistical test (e.g., t-test) to determine the significance of the difference in HDL cholesterol levels.
5. Interpret the results and quantify the effect size of smoking on HDL cholesterol levels.
'''
def test(dataset):
    import pandas as pd
    import numpy as np
    from scipy.stats import ttest_ind
    # Define the groups
    smokers = dataset[dataset['smoking_status'] == 1]
    non_smokers = dataset[dataset['smoking_status'] == 0]
    # Calculate the average HDL cholesterol levels
    avg_hdl_smokers = smokers['HDL'].mean()
    avg_hdl_non_smokers = non_smokers['HDL'].mean()
    # Perform t-test
    t_stat, p_val = ttest_ind(smokers['HDL'], non_smokers['HDL'])
    result = p_val < 0.05
    return result
insights = "Smokers have significantly lower levels of HDL cholesterol, with an average difference of 10 mg/dL compared to non-smokers. This highlights the potential cardiovascular risks associated with smoking, emphasizing the importance of smoking cessation programs and lipid profile monitoring for smokers."
