        return 1

        return 0

3. Calculate the mean price for diamonds with table values above 60 and those below 60.
4. Compare the mean prices between the two groups using a statistical test (e.g., t-test).

'''
def test(dataset):
    import numpy as np
    from scipy.stats import ttest_ind
    high_table_values = dataset[dataset['table'] > 60]
    low_table_values = dataset[dataset['table'] <= 60]
    high_table_prices = high_table_values['Unnamed: 0']
    low_table_prices = low_table_values['Unnamed: 0']
    mean_difference, p_value = ttest_ind(high_table_prices, low_table_prices)
    if mean_difference >= 2000 and p_value < 0.05:
        return 1
    else:
        return 0
