# Hypothesis 7.
description = "Cars manufactured after 2016 have a significantly lower rate of reported accidents compared to older models, reflecting advancements in vehicular safety technologies."
COT = '''
1. Define groups based on manufacturing year: vehicles post-2016 and pre-2016.
2. Calculate accident rates for both groups.
3. Conduct proportion tests (such as chi-squared tests) to assess if newer cars indeed have lower accident reports.
4. Set thresholds and significance levels to examine safety advancements.
5. Implement tests with functions validating accident impact based on manufacturing year.
'''
def test(dataset):
    from scipy.stats import chi2_contingency
    
    # Count accident-free cars manufactured after vs. before 2016.
    post_2016 = dataset[dataset['model_year'] > 2016]['accident'].value_counts()
    pre_2016  = dataset[dataset['model_year'] <= 2016]['accident'].value_counts()

    # Create contingency table for chi-squared test
    contingency_table = np.array([
        [post_2016.get('None reported', 0), post_2016.sum() - post_2016.get('None reported', 0)],
        [pre_2016.get('None reported', 0), pre_2016.sum() - pre_2016.get('None reported', 0)]
    ])

    # Chi-squared test
    chi2_stat, p_value, _, _ = chi2_contingency(contingency_table)
    
    result = int(p_value < 0.05)
    return result

insights = "The implementation of advanced safety features in vehicles manufactured post-2016 seems to have reduced accident rates significantly, reinforcing the value of newer models both in terms of human safety and insurance benefits."

def test(dataset):
    import pandas as pd
    from lifelines import CoxPHFitter

def test(dataset):
    import pandas as pd
    from sklearn.model_selection import train_test_split
    from sklearn.linear_model import LinearRegression
    
    accident_mapping = {
        "None reported": 0,
        "At least 1 accident or damage reported": 1,
        "2 or more accidents reported": 2 # Assuming this category
    }
    dataset['accident_count'] = dataset['accident'].map(accident_mapping).fillna(0)
    
    dataset['value_loss'] = dataset['price'] / (dataset['milage'] + 1)  # Avoid division by zero

    dataset_accident = dataset[dataset['accident_count'] >= 2]
    
    X_non = dataset_non_accident['milage'].values.reshape(-1, 1)
    y_non = dataset_non_accident['value_loss']
    
    X_acc = dataset_accident['milage'].values.reshape(-1, 1)
    y_acc = dataset_accident['value_loss']
    
    model_non = LinearRegression().fit(X_non, y_non)
    model_acc = LinearRegression().fit(X_acc, y_acc)
    
    coef_ratio = (model_acc.coef_ / model_non.coef_).item()
    result = int(coef_ratio >= 2)
    
    return result
