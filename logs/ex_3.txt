def test(dataset):
    import pandas as pd
    from sklearn.preprocessing import StandardScaler
    from sklearn.cluster import KMeans
    from sklearn.metrics import silhouette_score
    import numpy as np

    df = dataset.dropna()
    df = df.replace({'Sex': {'male': 0, 'female': 1}, 'Embarked': {'C': 0, 'Q': 1, 'S': 2}})
    features = df.drop(columns=['Name', 'Ticket', 'Cabin', 'Survived'])

    scaler = StandardScaler()
    features_scaled = scaler.fit_transform(features)

    # KMeans clustering with all features
    kmeans = KMeans(n_clusters=3)
    cluster_labels_full = kmeans.fit_predict(features_scaled)
    score_full = silhouette_score(features_scaled, cluster_labels_full)

    # KMeans clustering with a subset of features
    features_subset = df[['Pclass', 'Sex', 'Age', 'Fare']]
    features_subset_scaled = scaler.fit_transform(features_subset)
    kmeans_subset = KMeans(n_clusters=3)
    cluster_labels_subset = kmeans_subset.fit_predict(features_subset_scaled)
    score_subset = silhouette_score(features_subset_scaled, cluster_labels_subset)

    # Compare stability
    score_diff = np.abs(score_full - score_subset)
    result = int(score_diff < 0.1)  # Accept if stability (small score_diff) is present
    return result
