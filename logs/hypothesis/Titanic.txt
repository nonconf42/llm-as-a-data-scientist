def test(dataset):
    import pandas as pd
    import statsmodels.api as sm
    import numpy as np
    
    # Extract ticket prefix
    dataset['Ticket_Prefix'] = dataset['Ticket'].str.extract(r'(\D*)')[0]
    specific_prefix = dataset['Ticket_Prefix'].value_counts().idxmax()  # Focus on one frequent prefix
    
    X = sm.add_constant(pd.get_dummies(dataset['Ticket_Prefix'] == specific_prefix, drop_first=True))
    y = dataset['Survived']
    
    model = sm.Logit(y, X).fit(disp=False)
    p_val = model.pvalues.iloc[1]
    result = 1 if p_val < 0.05 else 0
    return result
