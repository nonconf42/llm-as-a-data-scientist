# Hypothesis 1.
description = "Class 1 passengers have a higher survival rate than other classes."
def test_1(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency

    # Pivot table for survival rates by class
    table = pd.crosstab(dataset["Pclass"], dataset["Survived"])
    # Chi-squared test
    chi2, p, dof, ex = chi2_contingency(table)

    # Use a significance level of 0.05
    result = 1 if p < 0.05 else 0
    return result

description = "Passengers traveling alone (SibSp == 0 and Parch == 0) had different survival rates compared to those not alone."
def test_8(dataset):
    import pandas as pd
    from scipy.stats import wilcoxon

    # Single passenger definition
    dataset["Is_Alone"] = (dataset["SibSp"] == 0) & (dataset["Parch"] == 0)

    # Pivot table
    table = pd.crosstab(dataset['Is_Alone'], dataset['Survived'])
    _, p = wilcoxon(table[0], table[1], alternative="less")

    result = 1 if p < 0.05 else 0
    return result

def test_10(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency

    # Pivot table for sex and class

    result = 1 if p < 0.05 else 0
    return result

# Hypothesis 11.
description = "Older passengers had a lower survival rate."
def test_11(dataset):
    import numpy as np
    from scipy import stats
    
    older_passengers = dataset[dataset['Age'] >= dataset['Age'].median()]['Survived']
    younger_passengers = dataset[dataset['Age'] < dataset['Age'].median()]['Survived']
    
    _, p = stats.ttest_ind(younger_passengers, older_passengers)

    result = 1 if p < 0.05 else 0
    return result

# Hypothesis 12.
description = "Fares are significantly different between passengers who survived and those who did not."
def test_12(dataset):
    from scipy.stats import ttest_ind

    survived = dataset[dataset['Survived'] == 1]['Fare'].dropna()
    not_survived = dataset[dataset['Survived'] == 0]['Fare'].dropna()

    _, p = ttest_ind(survived, not_survived)

    result = 1 if p < 0.05 else 0
    return result

# Hypothesis 13.
description = "First-class passengers have significantly different age distributions compared to other classes."
def test_13(dataset):
    from scipy.stats import kruskal

    first_class = dataset[dataset['Pclass'] == 1]['Age'].dropna()
    other_classes = dataset[dataset['Pclass'] != 1]['Age'].dropna()

    _, p = kruskal(first_class, other_classes)

    result = 1 if p < 0.05 else 0
    return result

# Hypothesis 14.
description = "Ticket number has no effect on survival rate."
def test_14(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency

    # Count unique tickets and survival
    ticket_survival = dataset.groupby('Ticket')['Survived'].nunique()
    
    # Creating a crosstab
    table = pd.crosstab(ticket_survival.index, ticket_survival)

    _, p, _, _ = chi2_contingency(table)

    result = 1 if p < 0.05 else 0
    return result

# Hypothesis 15.
description = "Passenger survival does not depend on the length of their name."
def test_15(dataset):
    from scipy.stats import ttest_ind
   
    # Calculate name length
    dataset['Name_Length'] = dataset['Name'].map(len)
    
    # Survived and not survived
    survived = dataset[dataset['Survived'] == 1]['Name_Length']
    not_survived = dataset[dataset['Survived'] == 0]['Name_Length']
    
    _, p = ttest_ind(survived, not_survived)

    result = 1 if p < 0.05 else 0
    return result
