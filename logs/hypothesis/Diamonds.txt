# Hypothesis 1.
insights = "Identifying clusters based on carat value may indicate market segments with different preferences and budget ranges."
description = "This hypothesis aims to identify natural clusters within the dataset based on the carat value of diamonds, reflecting distinct market segments."
COT = "To explore this hypothesis, employ the K-Means clustering algorithm on the 'carat' feature to identify if there are natural groupings. First, determine the optimal number of clusters using the Elbow Method. Then, apply K-Means clustering to identify potential clusters. Finally, interpret whether the identified clusters are significant and return the result."
def test(dataset):
    from sklearn.cluster import KMeans
    from sklearn.metrics import silhouette_score
    import numpy as np
    
    # Elbow Method to determine optimal number of clusters
    def elbow_method(data, max_clusters=10):
        distortions = []
        for i in range(1, max_clusters+1):
            km = KMeans(n_clusters=i, random_state=42)
            km.fit(data)
            distortions.append(km.inertia_)
        return distortions
    
    data = dataset[['carat']].values

    distortions = elbow_method(data)
    optimal_clusters = 1 + np.argmin(distortions[1:] - distortions[:-1])

    km = KMeans(n_clusters=optimal_clusters, random_state=42)
    labels = km.fit_predict(data)
    
    silhouette_avg = silhouette_score(data, labels)
    
    result = 1 if silhouette_avg > 0.5 else 0
    return result

def test(dataset):
    from sklearn.cluster import KMeans
    from sklearn.ensemble import RandomForestRegressor
    from sklearn.inspection import permutation_importance
    import numpy as np
    
    # Define clustering step
    def cluster_data(data, n_clusters):
        km = KMeans(n_clusters=n_clusters, random_state=42)
        clusters = km.fit_predict(data)
        return clusters

    n_clusters = 3
    data = dataset[['carat', 'cut', 'color', 'clarity', 'depth', 'table', 'x', 'y', 'z']].copy()
    categorical_cols = data.select_dtypes(include=['object']).columns

    data[categorical_cols] = data[categorical_cols].apply(lambda x: x.astype('category').cat.codes)
    
    clusters = cluster_data(data, n_clusters)
    
    # Define importance step
    def evaluate_feature_importance(data, clusters, target='price'):
        importances = []
        for cluster_id in np.unique(clusters):
            cluster_data = data[clusters == cluster_id]
            y = cluster_data[target]
            X = cluster_data.drop(columns=[target])
            model = RandomForestRegressor(n_estimators=100, random_state=42)
            model.fit(X, y)
            result = permutation_importance(model, X, y, n_repeats=10, random_state=42, n_jobs=2)
            importances.append(result.importances_mean)
        return np.array(importances).mean(axis=0)

    importances = evaluate_feature_importance(dataset, clusters)
    
    result = 1 if np.any(importances > (importances.mean() + importances.std())) else 0
    return result
