3. Perform a regression where Equity is dependent on ROA to examine relationship.
4. A negative and significant coefficient of ROA in the regression indicates validity.
5. Validate with a p-value under 0.05.
'''
def test(dataset):
    import statsmodels.api as sm
    
    roa = dataset['ROA']
    equity = dataset['Equity']
    
    # Linear regression analysis
    X = sm.add_constant(roa)
    model = sm.OLS(equity, X).fit()
    
    # Get p-value and coefficient
    p_value = model.pvalues[1]
    coeff = model.params[1]
    
    # Significance level
    alpha = 0.05
    
    # Test the hypothesis
    result = 1 if p_value < alpha and coeff < 0 else 0
    return result
