insights = "Identifying natural health indicator groupings could discover common profiles that either correlate with a higher or lower likelihood of smoking."
description = "Cluster Formation Hypothesis: Use k-means clustering to identify natural groupings within the dataset of health indicators and correlate them with the smoking status."
COT = "1. Select k-means clustering as the algorithm to use for cluster formation. 2. Determine the optimal number of clusters using the Elbow Method. 3. Apply k-means clustering on the dataset. 4. Analyze clusters' correspondence to smoking status using statistical methods like chi-square to see if a significant relationship exists."
def test(dataset):
    from sklearn.cluster import KMeans
    from sklearn.preprocessing import StandardScaler
    from sklearn.metrics import silhouette_score
    import numpy as np
    from scipy.stats import chi2_contingency

    # Standardizing data
    scaler = StandardScaler()

    # Determine the optimal number of clusters using the Elbow Method
    inertia = []
    range_n_clusters = range(2, 10)
    for n_clusters in range_n_clusters:
        kmeans = KMeans(n_clusters=n_clusters, random_state=42).fit(data_scaled)
        inertia.append(kmeans.inertia_)

    optimal_clusters = np.argmin(inertia) + 2  # Based on Elbow, assume optimal at min inertia
    kmeans = KMeans(n_clusters=optimal_clusters, random_state=42)
    dataset['Cluster'] = kmeans.fit_predict(data_scaled)

    # Analyze correspondence of clusters with smoking status using chi-square test
    contingency_table = pd.crosstab(dataset['Cluster'], dataset['smoking_status'])
    chi2, p, _, _ = chi2_contingency(contingency_table)

    result = 1 if p < 0.05 else 0
    return result
