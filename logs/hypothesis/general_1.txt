# Hypothesis 1.
description = "The survival rate differs significantly across different ticket classes."
def test(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency
    
    # Create a contingency table
    contingency_table = pd.crosstab(dataset['Pclass'], dataset['Survived'])
    chi2, p, dof, expected = chi2_contingency(contingency_table)
    
    # If p-value is less than 0.05, we reject the null hypothesis
    result = int(p < 0.05)
    return result

    import numpy as np

description = "The number of siblings/spouses on board affects survival more than the number of parents/children."
def test(dataset):
    from scipy.stats import logistic
    import statsmodels.api as sm

    # Prepare the data
    X = dataset[['SibSp', 'Parch']]
    y = dataset['Survived']
    X = sm.add_constant(X) # Adds a constant term to the predictor

    # Fit the logistic regression model
    model = sm.Logit(y, X).fit()
    p_values = model.pvalues
    
    # We accept the hypothesis if SibSp has a lower p-value than Parch in predicting survival
    result = int(p_values['SibSp'] < p_values['Parch'])
    return result

description = "Survivors were generally younger than those who did not survive."
def test(dataset):
    from scipy.stats import wilcoxon
    
    # Using Wilcoxon test to compare medians
    age_survived = dataset[dataset['Survived'] == 1]['Age'].dropna()
    age_not_survived = dataset[dataset['Survived'] == 0]['Age'].dropna()
    
    # The Wilcoxon test requires the two samples to be of equal length, remove items from larger sample
    min_len = min(len(age_survived), len(age_not_survived))
    stat, p = wilcoxon(age_survived[:min_len], age_not_survived[:min_len])
    
    result = int(p < 0.05)
    return result
