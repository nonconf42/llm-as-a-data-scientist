# Hypothesis 1.
description = "Passengers in higher classes (Pclass) had a higher survival rate."
def test1(dataset):
    import pandas as pd
    from scipy.stats import chi2_contingency

    # Create a contingency table of survival vs Pclass
    contingency_table = pd.crosstab(dataset['Pclass'], dataset['Survived'])
    # Perform the Chi-square test
    chi2, p, _, _ = chi2_contingency(contingency_table)
    
    # Check if the p-value is less than 0.05 (typical significance level)
    result = int(p < 0.05)
    return result

description = "There is a significant difference in mean age across different Pclass groups."
def test11(dataset):
    import numpy as np
    from scipy.stats import f_oneway

    # Drop missing ages
    dataset = dataset.dropna(subset=['Age'])

    # Separate ages by Pclass
    age_pclass1 = dataset[dataset['Pclass'] == 1]['Age']
    age_pclass2 = dataset[dataset['Pclass'] == 2]['Age']
    age_pclass3 = dataset[dataset['Pclass'] == 3]['Age']
    
    # Perform ANOVA
    f_stat, p = f_oneway(age_pclass1, age_pclass2, age_pclass3)
    
    result = int(p < 0.05)
    return result
