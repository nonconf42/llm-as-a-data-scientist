def test(dataset):
    import pandas as pd
    from scipy.stats import chisquare

    # Create a contingency table of Embarked and discretized Fare
    fare_labels = pd.qcut(dataset['Fare'].rank(method='first'), 5)  # Discretizing Fare into quintiles
    contingency_table = pd.crosstab(dataset['Embarked'], fare_labels)

    # Apply Chi-Square test
    chi2_stat, p_value, _, _ = chisquare(contingency_table.to_numpy())

    # Set significance threshold
    alpha = 0.05

    # Accept hypothesis if p_value is less than alpha
    result = int(p_value < alpha)
    return result
